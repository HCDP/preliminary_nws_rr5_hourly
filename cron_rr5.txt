# Sources envs/rr5.env first: the scripts REQUIRE RR5_PROJECT_ROOT and abort
# without it, and cron's environment inherits nothing from a shell.
# ${RR5_CODE_DIR:-...} keeps the line correct whether or not that optional
# override is set in the env file.
